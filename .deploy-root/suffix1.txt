/* ============ 核心算法 ============ */
const KEY=(t,a,v)=>t+'|'+a+(v?'|'+v:'');
/* 有「方案」维度的产品(如乐享丰年：保险期间×祝寿金)，费率/现价按 交费期间|投保年龄|方案 取数。
   cv_unit=基本保额：现价表为每1000元基本保额口径，按该组合费率(每1000元保费对应保额)折回每1000元保费口径，
   这样存档数据与官方表逐字一致，换算只在计算时发生。 */
function avail(pid,g,t,a,v){const k=KEY(t,a,v);const r=D.rate[pid]?.[g]?.[k];let c=D.cv[pid]?.[g]?.[k];
 if(!(r&&c))return null;
 if(byId[pid]?.cv_unit==='基本保额')c=c.map(x=>x*r/1000);
 return {rate:r,cv:c};}
/* 方案：产品的 variants 列表(code/label/可选性别)，当前选择存于 VAR；无 variants 的产品返回 undefined */
const VAR={};
function variantsOf(pid,g){const vs=byId[pid]?.variants;if(!vs)return null;
 return vs.filter(x=>!x.genders||x.genders.includes(g));}
function varOf(pid,g){const vs=variantsOf(pid,g);if(!vs)return undefined;
 if(!vs.some(x=>x.code===VAR[pid]))VAR[pid]=byId[pid].default_variant||vs[0].code;return VAR[pid];}
function varLabel(pid,v){return (byId[pid]?.variants||[]).find(x=>x.code===v)?.label||'';}

/* 责任规则驱动的取值：base含"保额"取SA，否则取当期累计已交保费cum */
function baseVal(base,SA,cum){return (base||'').includes('保额')?SA:cum;}

/* 年金/特别年金：从责任规则表逐条匹配当前交费期间(payment_term_filter留空=通用)、
   起领年度、周期(period_years，缺省1年)、末年是否排除(exclude_final_year)、
   以及age_threshold前后两档费率。同一年可能命中多条规则(如永裕的年金+特别年金)，累加。 */
/* 祝寿金(乐享丰年可选责任)：按到达年龄 start_age~end_age 给付，仅在所选方案含对应起领年龄(variant_filter)时生效；
   与年金分开返回，逐年表可单列，身故金扣减时一并计入已领。 */
function benefitOf(pid,n,a,last,t,SA,cum,v){
 const rules=(D.rules||[]).filter(r=>r.product_id===pid&&(r.rule_type==='年金'||r.rule_type==='特别年金'||r.rule_type==='祝寿金'));
 let amt=0,zs=0;
 for(const r of rules){
   const pf=r.payment_term_filter;
   if(pf!==''&&pf!=null&&+pf!==t)continue;
   if(r.variant_filter&&!(v||'').split('-').includes(r.variant_filter))continue;
   if(r.start_age&&a+n<+r.start_age)continue;
   if(r.end_age&&a+n>+r.end_age)continue;
   const sy=+r.start_year||1, period=+r.period_years||1;
   if(n<sy||(n-sy)%period!==0)continue;
   if(r.end_year&&n>+r.end_year)continue;
   if(r.exclude_final_year==='是'&&n===last)continue;
   const rate=(r.age_threshold&&(a+n)>+r.age_threshold)?+r.rate_after_threshold:+r.rate;
   if(r.rule_type==='祝寿金')zs+=rate*baseVal(r.base,SA,cum);else amt+=rate*baseVal(r.base,SA,cum);
 }
 return {ann:amt,zs};
}
/* 满期金：责任规则表里唯一一条"满期金"行决定计价基础(基本保额 或 累计已交保费)；比例为0＝无满期金(如乐享丰年) */
function maturityOf(pid,SA,cumFinal){
 const r=(D.rules||[]).find(x=>x.product_id===pid&&x.rule_type==='满期金');
 if(!r)return cumFinal;
 return (r.rate===''||r.rate==null?1:+r.rate)*baseVal(r.base,SA,cumFinal);
}
/* 现价表口径两种：
   含当期给付(友邦)：表列值已含当年年金/满期金，退保可得＝表列值；
   不含当期给付(中宏，cv_excludes_benefit=是)：表列值是周年日给付之后的现价，
   当日还会另外领到年金/满期金，故退保可得 cvS＝表列值＋当日年金＋(末年)满期金。
   逐年表"现金价值"列始终显示官方表列值，打平/IRR 一律用 cvS。 */
function cvExcl(pid){return byId[pid]?.cv_excludes_benefit==='是';}
/* 身故金两种口径：默认 max(现价含当日给付, 累计已交保费)；
   death_net_paid=是(乐享丰年)：max(现价含当日给付, 累计已交保费−此前已领生存年金与祝寿金)，与说明书演示逐行一致。 */
function project(pid,g,t,a,prem,div,v){
 const av=avail(pid,g,t,a,v); if(!av||!prem)return null;
 const k=prem/1000, SA=av.rate*k, last=av.cv.length-1, rows=[], ex=cvExcl(pid), dn=byId[pid]?.death_net_paid==='是';
 const mat=maturityOf(pid,SA,t===0?prem:Math.min(last,t)*prem);
 let cumA=0,cumD=0;
 for(let n=1;n<=last;n++){
   const cum=t===0?prem:Math.min(n,t)*prem;
   const b=benefitOf(pid,n,a,last,t,SA,cum,v), ann=b.ann+b.zs;
   const cvn=av.cv[n]*k, d=div?(div[n-1]||0):0;
   const fim=byId[pid]?.cv_final_includes_maturity==='是';
   const cvS=ex?cvn+ann+(n===last&&!fim?mat:0):cvn;
   const prevA=cumA; cumA+=ann; cumD+=d;
   rows.push({n,age:a+n,cum,ann,zs:b.zs,mat:n===last?mat:0,cumA,cv:cvn,cvS,div:d,cumD,
     death:dn?Math.max(cvS,cum-prevA):Math.max(cvS,cum),tot:cvS+cumA-ann,totD:cvS+cumA-ann+cumD});
 }
 return {SA,last,rows,term:t,prem,ex,v};
}
/* 现金流：t=0..term-1 交保费；t=5..n-1 领年金(+红利)；t=n 退保拿现价(含当年年金) */
function irr(pj,n,withDiv){
 const f=[],t=pj.term;
 for(let i=0;i<=n;i++)f.push(0);
 for(let i=0;i<(t===0?1:t);i++)if(i<=n)f[i]-=pj.prem;
 pj.rows.forEach(r=>{if(r.n<n)f[r.n]+=r.ann; if(withDiv&&r.n<=n)f[r.n]+=r.div;});
 f[n]+=pj.rows[n-1].cvS;
 const npv=x=>f.reduce((s,c,i)=>s+c/Math.pow(1+x,i),0);
 let lo=-0.9,hi=1;
 if(npv(lo)*npv(hi)>0)return null;

/* 持有到期口径：不退保，只看累计已领年金（末年另加满期金）是否覆盖累计已交保费 */
function beHold(pj,withDiv){
 for(const r of pj.rows){
   const got=r.cumA+r.mat+(withDiv?r.cumD:0);
   if(got>=r.cum)return r.n;}
 return null;
}
/* 退保口径 */
function breakeven(pj,withDiv){

   if(got>=r.cum)return r.n;}
 return null;
}

<h2>市场总览：快返年金现在是什么样子</h2>
<p class="note">下面这几条是把已入库的 9 款（4 家公司：友邦、中宏、大都会三家外资/合资，平安一家中资；约 59 万条现价记录）横着看出来的规律，不针对某一款产品。数值都是<b>保证利益</b>口径。</p>

<div class="card"><h2 style="margin:0 0 6px">一、「快返」快的是<b>时点</b>，不是<b>收益</b></h2>
<p>九款全部可以在<b>第 5~7 个保单周年日</b>开始给钱，最早的第 5 年，最晚的 5 年交也不过第 7 年，这条已经是行业标配，不构成差异。真正不同的是<b>那笔钱从哪来</b>，目前市场上有四种给法：</p>
<ul>
<li><b>按保费的固定比例给</b>（永裕两款 1.25%、宏瑞年年 2%、都会长盈 1.75%）：金额与投保年龄、性别无关，条款最好懂，客户一算就明白。</li>
<li><b>按基本保额的固定比例给</b>（恒曜、金生荣耀 20%，乐享丰年 10%）：金额取决于保额，而保额由年龄决定，同样的保费不同年龄拿到的钱差很多。</li>
<li><b>按基本保额全额给</b>（都会乐享、平安盛世金越 100%/年）：年金看着最厚，但保额本身被定价压得很低，本质是把本金按年摊还。平安还把<b>领取期间（9~29 年或至 105 岁）和起领年龄（最晚 90 岁）</b>做成可选，快返之外又能当定期养老金用。</li>
<li><b>阶段性加码</b>（恒曜/金生荣耀 64 周岁后翻倍、宏瑞年年与永裕两款的特别年金、乐享丰年的祝寿金、都会长盈交费期满的一次性特别生存金）：把一部分利益后置或集中到某个时点。</li>
</ul>
<p><b>洞察：</b>前两年的产品竞赛集中在「第几年开始领」，现在已经卷到头了；新上市的几款（盛世金越 2026-03、都会长盈 2026-06、永裕人生 2026-06、金生荣耀 2026-07、恒曜 2026-08）转而在<b>给付节奏的设计</b>上做文章。跟客户讲「快返」已经没有说服力，讲「这笔钱是本金还是本金之外的钱」才有。</p></div>

<div class="card"><h2 style="margin:0 0 6px">二、保证收益全部落在 1.4%~1.8% 这条窄带里</h2>
<p>九款主力方案的保证 IRR（持有到期）区间是 <b>1.42%~1.80%</b>，整体不到 0.4 个百分点。<b>不分红的两款</b>（宏瑞年年 1.80%、乐享丰年 1.68%）站在带子顶端，<b>分红的七款</b>保证部分只有 1.42%~1.60%（平安盛世金越趸交至 105 岁 1.47%），普遍比不分红的低 0.1~0.4 个百分点。</p>
<p>唯一跌出这条带子的是平安的<b>短领取期间方案</b>：领 9 年只有 0.85%、领 29 年 1.29%，官方说明书案例（6 年交、60 岁起领 19 年）保证 IRR 1.10%。领得越集中、越早领完，保证收益越低，这是用收益换「定期领完」。</p>
<p>这是预定利率下调之后的必然结果：保证部分已经没有多少腾挪空间，各家把差异化让渡给了红利。友邦四款说明书演示的含红利 IRR 在 <b>2.67%~2.76%</b>，平安说明书案例按当年红利口径约 2.2%，看起来拉开了一大截，但那是演示、不是承诺。</p>
<p><b>洞察：</b>在这条窄带里比小数点后两位意义不大——同一款产品换个交费期间，收益差异（最多 0.35 个百分点）就超过了产品之间的差异。<b>先把交费期间和投保年龄定下来，再比产品，顺序反了容易得出错误结论。</b></p></div>

<div class="card"><h2 style="margin:0 0 6px">三、真正拉开差距的是现金价值曲线，不是收益率</h2>
<p>九款的现价形态只有三种，决定了这笔钱中途能不能动、最后能不能留给下一代：</p>
<ul>
<li><b>本金留存型（6 款）</b>：现价长期稳在已交保费的 100% 上下，到期再返还一笔（按已交保费或按基本保额）。全程累计领取约为已交保费的 204%~220%。年金是本金之外多给的。</li>
<li><b>本金摊还型（3 款：乐享丰年、都会乐享、平安盛世金越）</b>：现价见顶后一路下滑，<b>没有满期金或只剩一个基本保额</b>（平安，约为保费的 2.6%），身故金也随已领年金递减。全程累计领取 108%~213%。年金里有很大一块本来就是客户自己的本金。同一形态里前期现价差别很大：平安第 8 年就能退保打平，都会乐享要第 17 年。</li>
<li><b>利益后置型（1 款：金生荣耀）</b>：虽然到期返还已交保费，但中途现价长期低于已交保费（第 10 年只有 66%），要到第 65 年才追平。</li>
</ul>
<p>可退保回本的年份从<b>第 5 年到第 23 年</b>，差了 18 年。同样是「保本型年金」，客户第 8 年想拿钱出来，有的产品全身而退，有的要亏一大截。</p>
<p><b>洞察：</b>收益率差 0.4 个百分点，回本时间差 18 年——<b>流动性的差异比收益的差异大一个数量级</b>。客户说不准将来会不会动这笔钱时，这一条应该比 IRR 先问。</p></div>

<div class="card"><h2 style="margin:0 0 6px">四、渠道决定了产品的复杂度</h2>
<p><b>银保三款</b>（永裕优享版、永裕人生、都会长盈）不约而同选了最简单的结构：年金按已交保费的固定比例给、到期把本金原样退回、没有可选责任、交费期最多 3 档。</p>
<p><b>个险六款</b>（恒曜、金生荣耀、宏瑞年年、乐享丰年、都会乐享、盛世金越）普遍更复杂：保额比例给付、64 周岁翻倍、祝寿金可选、保险期间或领取期间可选、交费期最多 6 档、年金还能改月领。</p>
<p><b>洞察：</b>这是渠道决定的——银保柜面讲解时间短，必须一句话说清；个险有代理人做方案，复杂结构才有空间。所以<b>拿银保产品和个险产品直接比 IRR 是不公平的</b>，它们面对的客户和销售场景不同。</p></div>

<div class="card"><h2 style="margin:0 0 6px">五、定价在趋同，条款在分化</h2>
<p>收益已经比不出花样，各家转而在规则上做区隔，这些往往才是最后成交与否的关键：</p>
<ul>
<li><b>投保年龄上限</b>：友邦四款、都会乐享、平安盛世金越 70 周岁，中宏两款 75 周岁，都会长盈 <b>80 周岁</b>。70 岁以上的客户，九款里只有三款能投。</li>
<li><b>减额交清</b>：九款里<b>只有宏瑞年年</b>可以。客户以后交不动保费时，其余八款只能退保或减保。</li>
<li><b>领取频率</b>：宏瑞年年、都会乐享、盛世金越可改月领（全年合计多 0.8%~2%），其余只能年领。</li>
<li><b>保险期间 / 领取期间可选</b>：乐享丰年能在 85 / 100 周岁之间选；平安盛世金越领取期间 6 档、起领年龄可约定到 90 岁，是九款里最灵活的；其余固定到 105 周岁。</li>
<li><b>第二投保人</b>：中宏两款可指定，用于投保人身故后的保单传承。</li>
<li><b>责任免除条数</b>：大都会两款各 3 条最宽松，友邦永裕两款和平安盛世金越 7 条最严（含毒品、酒驾等）。</li>
<li><b>信息披露</b>：外资/合资三家各交费期间的现价表都在官网全量公开；平安只公开趸交现价，期交现价要靠说明书案例或销售系统，同业比较时平安的期交方案只能看个别案例。</li>
</ul>
<p><b>洞察：</b>这些条款差异在利益演示表上一个数字都看不出来，却直接决定「这单能不能做、日后好不好服务」。<b>做方案时先用规则筛掉不可能的，再在剩下的里面比收益。</b></p></div>

<div class="card"><h2 style="margin:0 0 6px">六、两个容易被忽略的共性</h2>

<h2>总体规律</h2>

<div class="card"><h2 style="margin:0 0 6px">1. 先问客户一句话：要「稳稳的确定」还是「多一点可能」</h2>
<p>九款分两类。<b>不分红的</b>（宏瑞年年、乐享丰年心享版）合同写多少就是多少，保证 IRR 分别是 1.80% 和 1.68%。<b>分红的</b>（恒曜、金生荣耀、永裕两款、都会乐享、都会长盈、盛世金越）保证部分只有 1.42%~1.60%，低 0.1~0.4 个百分点，但多了一份红利：友邦四款说明书演示的到期 IRR 在 2.67%~2.76% 之间。</p>
<p>关键在于：<b>红利不保证</b>，分多少由公司经营结果决定，某些年份可能一分没有，演示数字不是承诺。所以这不是「谁收益高」，而是客户更想要哪一种。这一句讲清楚，比争论小数点后两位有用得多。</p>
<p class="src">友邦四个分红演示分别来自各自说明书的投保案例，年龄和交费期都不一样，只能看个大概量级，不能直接横比。大都会两款和平安说明书按「累积生息」演示红利且只列出部分年度，本工具已取其「当年度红利」列按直接领取口径建模（未列示年度线性插值），含分红数据可在「产品对比」页查看。</p></div>

<div class="card"><h2 style="margin:0 0 6px">2. 收益率差得很小，回本速度差得很大</h2>
<p>九款保证 IRR 基本挤在 1.42%~1.80% 这 0.4 个百分点里，可退保回本的年份却从第 5 年到第 23 年，差了四倍多：宏瑞年年 5 年、<b>都会长盈 5 年</b>、乐享丰年 6 年、永裕优享版 6 年、永裕人生 7 年、盛世金越 8 年（趸交）、恒曜 10 年、<b>都会乐享 20 年</b>、金生荣耀 23 年。</p>
<p>两款要特别留意：<b>金生荣耀</b>的现金价值长期低于已交保费，趸交要 21 年、20 年交要 29 年才回本，第 20 年退保算下来是 −0.40%（20 年交更差，−2.24%），也就是<b>亏本</b>；<b>都会乐享</b>的现金价值第 1 年只有已交保费的 66%，最高也就到 71%（第 4 年），之后一路下滑，靠每年领到的年金才慢慢追平，客户年纪越小、交费期越长回本越晚（0 岁趸交要到第 23 年，20 年交要到第 29 年）。</p>
<p><b>这是九款之间最实在的差别。</b>客户要是说不准以后会不会动这笔钱，这一条应该比 IRR 先问。</p></div>

<div class="card"><h2 style="margin:0 0 6px">3. 同样是「每年给钱」，来源完全不同</h2>
<p><b>本金留着的</b>（恒曜、金生荣耀、永裕两款、宏瑞年年、都会长盈）：到期还会再给一笔——前几款按已交保费、永裕两款按基本保额——全程累计领到的钱大约是已交保费的 204%~222%。年金是本金之外多给的。中途现金价值也一直在高位：永裕 104%~106%、宏瑞年年 100%、都会长盈 100%~102%、恒曜 99%~115%。</p>
<p><b>本金分批还的</b>（乐享丰年心享版、都会乐享、盛世金越）：现金价值最后归零或只剩一个基本保额，身故金也随着已领年金逐年变少。乐享丰年全程累计领取 144%~171%，都会乐享 132%~213%，盛世金越领到 105 岁 128%~211%、领 9 年只有 108%（岁数越小、领的年头越长，这个数越高）。这几款每年给的钱里，很大一块本来就是客户自己的本金。</p>
<p>所以这几款每年给得最多（乐享丰年例子里首年 8,729 元，是恒曜 3,060 元的 2.9 倍；都会乐享趸交 100 万每年领 26,890 元），但不能简单说「它给得多」。<b>要花掉的钱选它们，要留给下一代的本金别选。</b></p>
<p><b>金生荣耀是第三种</b>：到期虽然退还已交保费，但中途现金价值长期偏低（第 10 年 66%、第 30 年 87%、第 45 年 93%），要到第 65 年才追平。</p></div>

<div class="card"><h2 style="margin:0 0 6px">4. 钱什么时候到手，各家节奏不一样</h2>
<ul>
<li><b>都会乐享</b>：第 5 年起，每年给<b>一整个基本保额</b>（100%），一直给到 105 周岁。也可以改成月领，每月给基本保额的 8.4%，全年合计多 0.8%。</li>
<li><b>都会长盈</b>：交满后第 5 年（5 年交是第 6 年）先一次性给<b>一个基本保额</b>，此后每年给已交保费的 <b>1.75%</b>，一直给到期满，到期再全额退还已交保费。</li>
<li><b>乐享丰年</b>：第 5 年起每年给基本保额的 10%，金额固定，一直给到期满。加选祝寿金的话，再叠加每年已交保费的 5%，连给 20 年。</li>
<li><b>恒曜 / 金生荣耀</b>：第 5 年起给基本保额的 20%，<b>64 周岁后翻倍到 40%</b>。年轻时少、上年纪后多。</li>
<li><b>宏瑞年年</b>：每年给已交保费的 2%，另外在第 5~18 年（看交费期）连领 2 次特别年金，每次一个基本保额。</li>
<li><b>永裕两款</b>：每年给已交保费的 1.25%，另外从第 10 年起<b>每满 5 年</b>多给一次，每次是已交保费的 2.6%。</li>
<li><b>盛世金越</b>：每年给<b>一整个基本保额</b>（年领不超过年交保费×交费年数×20%），趸交最早第 5 年、期交最早第 6 年且交满后开始，最晚 90 周岁，<b>投保时约定起领年龄和领取期间</b>（9/14/19/24/29 年或至 105 周岁），领完再给一个基本保额满期金。可改月领（每月 8.4%）。</li>
</ul>
<p>到 70 周岁时一共领到手多少（占已交保费）：乐享丰年 76%、盛世金越 69%（趸交）、都会乐享 65%、宏瑞年年 50%、都会长盈 45%、永裕两款 44%、金生荣耀 34%、恒曜 33%。想「早点见到钱」，顺序很清楚。</p></div>

<div class="card"><h2 style="margin:0 0 6px">5. 客户年纪大，有五款不吃亏</h2>
<p>投保年龄从 0 岁提到 70 岁，保证 IRR 的降幅：永裕两款 −0.12（1.62%→1.50%）、宏瑞年年 −0.14（1.83%→1.69%）、都会乐享 −0.14（1.57%→1.43%，趸交口径）、盛世金越 −0.26（1.55%→1.29%，趸交至 105 岁）、乐享丰年 −0.73（1.77%→1.04%）；恒曜和金生荣耀的费率表只编到 70 岁，到 60 岁时已分别降到 1.31% 和 1.28%。</p>
<p>原因在算法：永裕的年金和满期金都按保费的固定比例给，宏瑞年年的年金也按已交保费算，都跟年龄无关；都会乐享则是岁数越大保额越高（同样 100 万趸交，40 岁每年领 26,890 元，60 岁领 34,680 元），刚好抵消了领取年头变短。盛世金越形态相同，但保额随年龄涨得没那么多（40 岁到 70 岁涨 57%，都会乐享涨 58% 且基数更高），高龄投保与都会乐享的差距拉大到 0.14 个百分点。乐享丰年是固定保到 85 或 100 周岁，岁数大了剩下的年头短，最吃亏。</p>
<p>都会长盈是另一种抗跌：0 岁到 80 岁保证 IRR 只在 1.63%~1.57% 之间，且<b>投保上限 80 周岁，是九款里最高的</b>。</p>
<p><b>55 岁以上的客户</b>优先看永裕两款、宏瑞年年、都会长盈和都会乐享；<b>给孩子买</b>则乐享丰年（1.77%~1.82%）和宏瑞年年（1.83%）领先——都会乐享给孩子买回本太慢（0 岁趸交要第 23 年）。</p></div>

<div class="card"><h2 style="margin:0 0 6px">6. 想分期慢慢交，有两家不打折</h2>
<p>从趸交改成最长的交费期，保证 IRR 的变化：永裕两款 0（趸交、3 年、5 年都是 1.59%）、宏瑞年年 −0.04、恒曜 −0.05、乐享丰年 −0.15、金生荣耀 −0.22（1.68%→1.46%）。<b>都会乐享 −0.35</b>（趸交 1.53%→20 年交 1.18%，可测算的八款里最敏感）；都会长盈趸交、3 年交、5 年交几乎持平（1.60% 上下）。平安盛世金越期交现价未公开，无法测算；说明书案例 6 年交、60 岁起领 19 年的保证 IRR 为 1.10%。</p>
<p>客户想拉长交费期，永裕和宏瑞年年基本不吃亏；金生荣耀选长交费期要特别谨慎，再叠加它本来就慢的回本速度，20 年交要等到第 29 年才回本。</p></div>

<div class="card"><h2 style="margin:0 0 6px">7. 七款男女同价，大都会两款分男女</h2>
<p>友邦四款、中宏两款和平安盛世金越在相同条件下男女的保证 IRR 完全一样（差异不到 0.01 个百分点）。乐享丰年虽然费率表分男女编，但差别只在祝寿金能从几岁开始领（女 55/60 周岁、男 60/65 周岁）。</p>
<p><b>大都会两款是例外</b>：同样的保费，女性拿到的基本保额略低（都会乐享 40 周岁趸交 100 万，男 26,890 元、女 26,730 元），保证 IRR 低 0.01~0.03 个百分点。因为女性预期领取的年头更长。给女性客户做方案时记得单独算一遍。</p></div>

<h2>每款产品小结</h2>
<p class="note">下面每款的数字都基于同一个例子：男 40 周岁、3 年交、每年 10 万元（平安盛世金越为趸交 30 万）。</p>

<div class="card"><h2 style="margin:0 0 4px">中宏宏瑞年年年金保险</h2>
<p class="src">普通型（不分红）· 个险代理人 · 2026-01-07 上市 · 保到 105 周岁 · 投保上限 75 周岁</p>
<p><b>怎么给钱：</b>每年给已交保费的 2%（6,000 元），趸交和 3 年交从第 7 个保单周年日开始，交费期越长开始越晚。另外有两次特别生存年金，每次等于一个基本保额。到期退还全部已交保费。</p>
<p><b>好在哪：</b>九款里保证 IRR 最高（1.80%），回本最快（第 5 年），投保年龄从 0 岁到 70 岁只差 0.14 个百分点。<b>只有它可以减额交清</b>——客户以后交不动保费时，这是唯一的缓冲。年金还能选按月领（全年合计多 2%）。</p>
<p><b>要提醒：</b>不分红，1.80% 就是上限，没有红利的想象空间。</p>
<p><b>适合：</b>只认确定数字、可能中途要用钱、或者年纪偏大的客户。</p></div>

<div class="card"><h2 style="margin:0 0 4px">中宏乐享丰年心享版年金保险</h2>
<p class="src">普通型（不分红）· 个险代理人 · 2025-09-01 上市 · 保到 85 或 100 周岁（投保时选）· 投保上限 75 周岁</p>
<p><b>怎么给钱：</b>第 5 年起每年给基本保额的 10%，一直到期满。可以加选祝寿金，从女 55/60、男 60/65 周岁起，每年再给已交保费的 5%，连给 20 年。<b>到期没有满期金</b>。</p>
<p><b>好在哪：</b>现金流最厚，到 70 周岁已经领回已交保费的 76%~91%。保证 IRR 1.60%~1.71%，在分红产品的保证部分之上。六种方案（保险期间 × 祝寿金）可以按客户的领钱节奏挑。前期现价也不低，第 6 年就能回本。</p>
<p><b>要提醒：</b>现金价值最后归零，身故金也逐年变少——它是把本金分批还给你，不是本金之外多给。客户年纪越大越不划算（70 岁投保只有 1.04%）。</p>
<p><b>适合：</b>要一笔确定的、能花掉的现金流，且不打算把这笔钱留给下一代的客户。</p></div>

<div class="card"><h2 style="margin:0 0 4px">友邦永裕人生优享版年金保险（分红型）</h2>
<p class="src">分红型 · 银保 · 2026-08-12 上市 · 保到 105 周岁 · 投保上限 70 周岁</p>
<p><b>怎么给钱：</b>每年给已交保费的 1.25%（3,750 元），趸交第 5 年、3 年交第 6 年、5 年交第 7 年开始。第 10 年起每满 5 年多给一次，每次是已交保费的 2.6%（13,000 元）。到期给一个基本保额（例子里 309,300 元）。</p>
<p><b>好在哪：</b>分红六款里保证部分最高（1.59%）、回本最快（第 6 年），而且从 0 岁到 70 岁只降 0.12 个百分点，交费期拉长也完全不减收益。说明书演示的含红利 IRR 是 2.76%，友邦四款里最高。</p>
<p><b>要提醒：</b>只能年交，没有半年、季、月交；不可减额交清。红利不保证。</p>
<p><b>适合：</b>想要红利空间、又不愿意在保证部分让步太多的客户；银行渠道客户。</p></div>

<div class="card"><h2 style="margin:0 0 4px">友邦永裕人生年金保险（分红型）</h2>
<p class="src">分红型 · 银保 · 2026-06-01 上市 · 保到 105 周岁 · 投保上限 70 周岁</p>
<p><b>怎么给钱：</b>和优享版<b>完全相同</b>——1.25% 的年金、每 5 年一次 2.6% 的特别年金、到期给一个基本保额、保到 105 周岁、身故金取现价与已交保费的较大者。</p>
<p><b>和优享版差在哪：</b>两点，都很小。一是费率微调，同样交 10 万，有的年龄新版保额略高、有的略低，平均相差不到 0.05%；二是前期现金价值，例子里第 7 年优享版已到 510,500 元（超过已交保费），这款是 409,300 元，要到第 8 年才超过，<b>回本晚一年</b>。</p>
<p><b>结论：</b>条件相同时优享版略占优。这一款比优享版早两个多月上市，同为银保渠道，利益上几乎没有损失。</p></div>

<div class="card"><h2 style="margin:0 0 4px">友邦传世金生2026荣耀版年金保险（分红型）</h2>
<p class="src">分红型 · 个险代理人 · 2026-07-01 上市 · 保到 105 周岁 · 投保上限 70 周岁</p>
<p><b>怎么给钱：</b>第 5 年起每年给基本保额的 20%（3,190 元），<b>64 周岁后翻倍到 40%</b>。到期退还全部已交保费。</p>
<p><b>好在哪：</b>交费期档位最全（趸交到 20 年交共 6 档）。长期持有到期，保证 IRR 1.54%，略高于恒曜；说明书演示含红利 2.70%。</p>
<p><b>要提醒：</b><b>这是九款里中途退保最吃亏的一款。</b>现金价值长期低于已交保费，第 20 年退保是 −0.40%，20 年交更是 −2.24%，要到第 23 年（趸交 21 年、20 年交 29 年）才回本。只适合确定几十年不动的钱。条款里写明红利来源仅为利差。</p>
<p><b>适合：</b>确定长期不动用、且认可红利逻辑的客户。凡是有中途退保可能的，换恒曜。</p></div>

const SUMBASE={g:'男',t:5,a:40,prem:100000};

function sumRows(F){
 const out=[],na=[];
 F.forEach(p=>{
  const pid=p.product_id, vs=variantsOf(pid,SUMBASE.g), codes=vs?vs.map(x=>x.code):[undefined];
  let any=false;
  codes.forEach(v=>{
   const pj=project(pid,SUMBASE.g,SUMBASE.t,SUMBASE.a,SUMBASE.prem||0,null,v);
   if(!pj)return; any=true;
   const FR=pj.rows[pj.last-1], a1=pj.rows.find(r=>r.ann>0);
   const n70=70-SUMBASE.a, r70=(n70>=1&&n70<=pj.last)?pj.rows[n70-1]:null;
   /* 峰值用纯现金价值(不含当日年金)：这一列回答的是「中途退保能拿回多少」 */
   let peak=0,peakN=0; pj.rows.forEach(r=>{if(r.cv/r.cum>peak){peak=r.cv/r.cum;peakN=r.n;}});
   const term=byId[pid]?.policy_terms||'';
   const ty=(term.match(/(\d+)\s*周岁/)||[])[1];
   out.push({p,v,pj,term:(v&&varLabel(pid,v).split('·')[0].trim())||(ty?'至'+ty+'周岁':''),
     SA:pj.SA,annN:a1?a1.n:null,ann1:a1?a1.ann:0,mat:FR.mat,
     peak,peakN,be:breakeven(pj,false),got70:r70?(r70.cumA+(r70.n===pj.last?r70.mat:0))/r70.cum:null,
     irr:irr(pj,pj.last,false),endAge:SUMBASE.a+pj.last});
  });
  if(!any)na.push(p.short_name||p.product_name);
 });
 out.sort((x,y)=>(y.irr==null?-9:y.irr)-(x.irr==null?-9:x.irr));
 return {out,na};
}

   中宏乐享丰年心享版自第5个保单周年日起按基本保额的10%年年领生存年金至期满日(含)，<b>无满期金</b>；可选祝寿金按累计已交保费的5%领20年(女55/60、男60/65周岁起)；身故金为现价与「累计已交保费−已领生存年金与祝寿金」的较大者；其费率与现价随保险期间(至85/至100周岁)和祝寿金选择而不同，在上方「方案」中切换。
   宏瑞年年按年领口径计算；月领(每月为年领额×8.5%，全年合计多2%，且逐月到账)未建模。<br>
   <b>现价峰值占累计保费</b> = 纯现金价值(不含当日年金)的最高点与对应的保单年度，回答的是「中途退保最多能拿回多少」。<br>
   <b>打平年度 · 退保口径</b> = 第一个满足「当年现价 + 起领年度至<u>上一</u>年度已领年金（+ 累计至当年的现金红利）≥ 累计已交保费」的保单年度。
   友邦现价表列示值已含当年度年金/特别年金与末年满期金，故已领部分只累加至上一年度，避免重复计入；中宏现价表列示值不含当日给付，本工具先把当日年金(末年加满期金)补进"当年现价"再套同一公式，两种口径结果可直接比较；现金红利不在保证利益现价内，累计至当年。<br>
   <b>打平年度 · 持有到期口径</b> = 不退保，第一个满足「累计已领年金/特别年金（含当年，末年另加满期金）（+ 累计现金红利）≥ 累计已交保费」的保单年度。此口径不动用现金价值。<br>
   <b>IRR</b> = 交费期内每年年初交保费为流出，起领年度起每年末领年金（按各产品规则，可能含特别年金）为流入，评估年度末退保领取现金价值为最后一笔流入（即退保口径）。<br>
   <b>含分红口径</b> = 现金红利型产品，按<b>当年红利直接领取现金、不累积生息</b>计算。友邦说明书即按此口径演示（其累计红利列为逐年简单累加，已核对一致）；大都会说明书按累积生息演示，本工具取其「当年度红利」列按直接领取建模，累计红利为简单累加、不含利息，低于说明书「累积红利」列。
   若实际选择「储存于本公司生息」或「抵缴保险费」，现金流时点不同，IRR 会高于此处数值；增额红利型产品的红利计入保额而非现金，需另行建模，不适用本口径。<br>
   <b>说明书演示段</b>的打平年度与 IRR 口径同上；累计现金红利由逐年演示值累加得出，与说明书累计列可能有几元的取整差异。</div></details>
  ${anyDemo||!items.some(x=>x.p.dividend_method!=='不分红')?'':'<div class="info">这套投保条件没有对应的官方利益演示，因此「利益试算」段只能给出保证利益，无法计算含分红 IRR。分红折算率公司未公开，任何非官方的红利推算都不可靠；含分红数据可看上表「说明书演示」段各产品的官方案例。</div>'}
  ${ok.length?chart(ok):''}`;
 if(CMPH)transposeCmp();
 renderYears();
}
function yearTable(x,i,withDiv){
 const r=x.pj.rows,F=r[r.length-1],p=x.p,vl=x.pj.v?varLabel(p.product_id,x.pj.v):'';
 const hasZ=r.some(y=>y.zs>0);
 const rowNote=p.demo_same_row==='是'
   ?'说明书演示同样把第 N 个保单周年日给付的年金列在第 N 行，行号可直接对照。'
   :'说明书演示把第 N 个保单周年日给付的年金列在第 N+1 行（年度初），本表列在第 N 行（年度末），时点相同只是行号差一。';
 const deathNote=p.death_net_paid==='是'
   ?'身故保险金列同说明书演示口径：周年日现价（含当日给付）与「累计已交保费 − 此前已领生存年金'+(hasZ?'与祝寿金':'')+'」取较大者。'
   :'身故保险金列同说明书演示口径：周年日现价（含当日给付）与累计已交保费取较大者。';
 const endNote=F.mat
   ?`期满日现价为 0，期满金 ${fmt(F.mat)} 与当年年金另付，合计 ${fmt(F.tot)}，为累计已交保费的 ${(F.tot/F.cum*100).toFixed(1)}%。`
   :`期满日现价为 0，<b>无满期金</b>，只领当年年金 ${fmt(F.ann)}；全期累计已领年金${hasZ?'与祝寿金':''} ${fmt(F.tot)}，为累计已交保费的 ${(F.tot/F.cum*100).toFixed(1)}%。`;
 return `<details class="card"><summary style="color:${CLR[i]}">${esc(p.product_name)}${vl?'（'+esc(vl)+'）':''} · 逐年利益</summary>
 <div class="scroll lock"><table><tr><th class="n">保单年度</th><th class="n">年龄</th><th class="n">累计已交保费</th>
  <th class="n">当年年金</th><th class="n">累计已领年金</th><th class="n">现金价值</th><th class="n">身故保险金</th>
  <th class="n">现价+已领年金</th>${withDiv?'<th class="n">当年现金红利</th><th class="n">累计现金红利</th><th class="n">合计（含红利）</th>':''}</tr>
 ${r.map(y=>`<tr><td class="n">${y.n}</td><td class="n">${y.age}</td><td class="n">${fmt(y.cum)}</td>
  <td class="n">${y.ann?fmt(y.ann):'—'}${y.zs?'<br><span class="src">含祝寿金 '+fmt(y.zs)+'</span>':''}${y.mat?'<br><span class="src">+满期 '+fmt(y.mat)+'</span>':''}</td>
  <td class="n">${fmt(y.cumA)}</td><td class="n">${fmt(y.cv)}</td><td class="n">${fmt(y.death)}</td>
  <td class="n">${fmt(y.tot)}</td>
  ${withDiv?`<td class="n">${fmt(y.div)}</td><td class="n">${fmt(y.cumD)}</td><td class="n">${fmt(y.totD)}</td>`:''}</tr>`).join('')}
 </table></div>
 ${x.pj.ex?`<p class="src">该产品现价表列值<b>不含</b>当个周年日给付的年金${hasZ?'、祝寿金':''}与满期金，本表"当年年金"与"现金价值"并列即为同一周年日的两笔钱；
  "现价+已领年金"已把当日年金${hasZ?'（含祝寿金）':''}计入。${rowNote}
  ${deathNote}
  ${endNote}</p>`
 :`<p class="src">末年（第 ${F.n} 保单年度）现价 ${fmt(F.cv)}，为累计已交保费的 ${(F.cv/F.cum*100).toFixed(1)}%；连同累计已领年金合计 ${fmt(F.tot)}，为 ${(F.tot/F.cum*100).toFixed(1)}%。</p>`}</details>`;
}
function chart(items){
 const W=700,H=250,L=62,R=12,T=10,B=24;
 const maxN=Math.max(...items.map(x=>x.pj.last));
 const max=Math.max(...items.flatMap(x=>x.pj.rows.map(r=>Math.max(r.tot,r.cum))));
 const px=n=>L+(n-1)/(maxN-1)*(W-L-R), py=v=>T+(1-v/max)*(H-T-B);
 const path=(rows,f,c,w,dash='')=>`<path fill="none" stroke="${c}" stroke-width="${w}" ${dash}
   d="${rows.map((r,i)=>(i?'L':'M')+px(r.n).toFixed(1)+' '+py(f(r)).toFixed(1)).join(' ')}"/>`;

const CRC_T=(()=>{const t=new Uint32Array(256);for(let n=0;n<256;n++){let c=n;for(let k=0;k<8;k++)c=c&1?0xEDB88320^(c>>>1):c>>>1;t[n]=c>>>0;}return t;})();
function crc32(u){let c=0xFFFFFFFF;for(let i=0;i<u.length;i++)c=CRC_T[(c^u[i])&255]^(c>>>8);return (c^0xFFFFFFFF)>>>0;}
function zipStore(files){ /* files: [{name, data:Uint8Array}] */
 const enc=new TextEncoder(), parts=[], cen=[]; let off=0;
 const le=(n,b)=>{const a=new Uint8Array(b);for(let i=0;i<b;i++)a[i]=(n>>>(8*i))&255;return a;};
 for(const f of files){
  const nm=enc.encode(f.name), crc=crc32(f.data), sz=f.data.length;
  const head=[le(0x04034b50,4),le(20,2),le(0x0800,2),le(0,2),le(0,2),le(0x21,2),le(crc,4),le(sz,4),le(sz,4),le(nm.length,2),le(0,2),nm];
  const hl=head.reduce((s,a)=>s+a.length,0);
  parts.push(...head,f.data);
  cen.push(le(0x02014b50,4),le(20,2),le(20,2),le(0x0800,2),le(0,2),le(0,2),le(0x21,2),le(crc,4),le(sz,4),le(sz,4),
   le(nm.length,2),le(0,2),le(0,2),le(0,2),le(0,2),le(0,4),le(off,4),nm);
  off+=hl+sz;}
 const cl=cen.reduce((s,a)=>s+a.length,0);
 const end=[le(0x06054b50,4),le(0,2),le(0,2),le(files.length,2),le(files.length,2),le(cl,4),le(off,4),le(0,2)];
 return new Blob([...parts,...cen,...end],{type:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'});}
const xesc=s=>String(s).replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c])).replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g,'');
function colName(i){let s='';i++;while(i){const m=(i-1)%26;s=String.fromCharCode(65+m)+s;i=Math.floor((i-1)/26);}return s;}
