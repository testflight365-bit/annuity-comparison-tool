function benefitOf(pid,n,a,last,t,SA,cum,v){

/* 满期金：责任规则表里唯一一条"满期金"行决定计价基础(基本保额 或 累计已交保费)；比例为0＝无满期金(如乐享丰年) */

/* 现价表口径两种：

   不含当期给付(中宏，cv_excludes_benefit=是)：表列值是周年日给付之后的现价，
   当日还会另外领到年金/满期金，故退保可得 cvS＝表列值＋当日年金＋(末年)满期金。

/* 现金流：t=0..term-1 交保费；t=5..n-1 领年金(+红利)；t=n 退保拿现价(含当年年金) */

}

const CRC_T=(()=>{const t=new Uint32Array(256);for(let n=0;n<256;n++){let c=n;for(let k=0;k<8;k++)c=c&1?0xEDB88320^(c>>>1):c>>>1;t[n]=c>>>0;}return t;})();

function zipStore(files){ /* files: [{name, data:Uint8Array}] */

 const le=(n,b)=>{const a=new Uint8Array(b);for(let i=0;i<b;i++)a[i]=(n>>>(8*i))&255;return a;};
 for(const f of files){
  const nm=enc.encode(f.name), crc=crc32(f.data), sz=f.data.length;
  const head=[le(0x04034b50,4),le(20,2),le(0x0800,2),le(0,2),le(0,2),le(0x21,2),le(crc,4),le(sz,4),le(sz,4),le(nm.length,2),le(0,2),nm];
  const hl=head.reduce((s,a)=>s+a.length,0);
  parts.push(...head,f.data);
  cen.push(le(0x02014b50,4),le(20,2),le(20,2),le(0x0800,2),le(0,2),le(0,2),le(0x21,2),le(crc,4),le(sz,4),le(sz,4),
   le(nm.length,2),le(0,2),le(0,2),le(0,2),le(0,2),le(0,4),le(off,4),nm);

function initUI(){
 const q=x=>document.getElementById(x);
 const cos=[...new Set(P.map(p=>p.company_code))], chs=[...new Set(P.map(p=>p.channel))];
 q('f-co').innerHTML='<option value="">全部</option>'+cos.map(c=>`<option value="${c}">${esc(CO[c]||c)}</option>`).join('');
 q('f-ch').innerHTML='<option value="">全部</option>'+chs.map(c=>`<option value="${esc(c)}">${esc(c)}</option>`).join('');
 const maxA=Math.max(...P.map(p=>+p.entry_age_max_years));
 q('c-a').innerHTML=Array.from({length:maxA+1},(_,i)=>`<option value="${i}">${i} 周岁</option>`).join('');
 q('c-t').innerHTML=ALLT.map(t=>`<button data-t="${t}">${tTxt(t)}</button>`).join('');
 q('meta').textContent=`${P.length} 款产品 · ${new Set(P.map(p=>p.company_code)).size} 家公司 · 数据更新至 ${UPDATED} · 数值由官方费率表与现价表还原`;
 sync();
 ['co','ch','des','own'].forEach(k=>q('f-'+k).onchange=e=>{S[k]=e.target.value;render();});
 q('f-q').oninput=e=>{S.q=e.target.value;render();};
 q('c-g').onchange=e=>{S.g=e.target.value;render();};
 q('c-a').onchange=e=>{S.a=+e.target.value;render();};
 q('c-t').querySelectorAll('button').forEach(b=>b.onclick=()=>{const t=+b.dataset.t;
  if(S.terms.includes(t)){if(S.terms.length>1)S.terms=S.terms.filter(x=>x!==t);}else S.terms=[...S.terms,t].sort((a,b)=>a-b);
  if(SORT.k&&!cols().some(c=>c.k===SORT.k))SORT.k=null;sync();render();});
 q('v-mode').querySelectorAll('button').forEach(b=>b.onclick=()=>{S.view=b.dataset.v;sync();render();q('tbl').scrollLeft=0;q('tbl').scrollTop=0;});
 q('v-shape').querySelectorAll('button').forEach(b=>b.onclick=()=>{S.shape=b.dataset.s==='1';
  if(SORT.k&&!cols().some(c=>c.k===SORT.k))SORT.k=null;sync();render();q('tbl').scrollLeft=0;});
 q('only').onclick=()=>{if(!PICK.size){alert('先在产品名前勾选要对比的产品');return;}ONLY=!ONLY;render();};
 q('clr').onclick=()=>{PICK.clear();ONLY=false;render();};
 q('reset').onclick=()=>{S=JSON.parse(JSON.stringify(DEF));PICK.clear();ONLY=false;SORT={k:null,d:1};
  for(const k in VAR)delete VAR[k];q('f-q').value='';sync();render();};
 q('xls').onclick=exportXlsx;
 render();}
function sync(){const q=x=>document.getElementById(x);
 q('f-co').value=S.co;q('f-ch').value=S.ch;q('f-des').value=S.des;q('f-own').value=S.own;q('c-g').value=S.g;q('c-a').value=S.a;
 q('c-t').querySelectorAll('button').forEach(b=>b.setAttribute('aria-pressed',S.terms.includes(+b.dataset.t)));
 q('v-mode').querySelectorAll('button').forEach(b=>b.setAttribute('aria-pressed',b.dataset.v===S.view));
 q('v-shape').querySelectorAll('button').forEach(b=>b.setAttribute('aria-pressed',(b.dataset.s==='1')===S.shape));}
initUI();
</script></body></html>
